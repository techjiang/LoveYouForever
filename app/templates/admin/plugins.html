怎么自己写一个插件？

from app.plugins.base import Plugin

class MyPlugin(Plugin):
    name = "my_plugin"
    title = "我的插件"
    version = "1.0.0"
    description = "做点有意思的事"

    def footer_html(self):
        return "<span>来自我的插件</span>"

    def on_post_render(self, post, html):
        return html  # 可以在这里二次加工文章内容

PLUGIN = MyPlugin()

可用钩子：on_install / on_enable / on_disable / on_uninstall / register_blueprint / before_request / after_request / on_post_render / nav_items / footer_html / dashboard_cards
